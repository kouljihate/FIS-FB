{# Reusable bilingual building blocks shared by every screen. The application chrome is bilingual everywhere: English and Arabic are always printed together and neither is ever hidden. Two arrangements are used: * ``bilingual_stack`` - Arabic on top, English below. This is the rule for tabs, buttons and table headers. * ``bilingual_heading`` / ``bilingual_label`` - English on the left, Arabic on the right. This is the rule for titles, section headings and form labels. #} {# Arabic on top, English below - tabs, buttons and table headers. #} {% macro bilingual_stack(en, ar, extra_class='') -%} {{ ar }} {{ en }} {%- endmacro %} {# Table header cell carrying both languages, Arabic first. #} {% macro th(en, ar, extra_class='') -%} {{ bilingual_stack(en, ar) }} {%- endmacro %} {# Shorthand: the text comes from the i18n table. #} {% macro bi_stack(key) -%} {{ bilingual_stack(t_en(key), t_ar(key)) }} {%- endmacro %} {% macro bi_label(key) -%} {{ bilingual_label(t_en(key), t_ar(key)) }} {%- endmacro %} {% macro bi_th(key, extra_class='') -%} {{ bilingual_stack(t_en(key), t_ar(key)) }} {%- endmacro %} {# English on the left, Arabic on the right - titles and headings. #} {% macro bilingual_heading(en, ar) -%} {{ en }} {{ ar }} {%- endmacro %} {# English on the left, Arabic on the right - form labels. #} {% macro bilingual_label(en, ar) -%} {{ en }} {{ ar }} {%- endmacro %} {# A bilingual button: Arabic above English, with an optional icon to the side. #} {% macro bilingual_button(en, ar, icon=None) -%} {{ ar }} {{ en }} {% if icon %}{% endif %} {%- endmacro %} {# Button text straight from the i18n table, both languages. #} {% macro bi_button(key, icon=None) -%} {{ bilingual_button(t_en(key), t_ar(key), icon) }} {%- endmacro %} {# Statistic tile. #} {% macro stat_tile(key_en, key_ar, value) -%}
